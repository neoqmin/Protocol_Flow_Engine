cmake_minimum_required(VERSION 3.16)
project(ProtocolFlowEngine LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(PF_BUILD_TESTS "Build tests" ON)
option(PF_WARNINGS_AS_ERRORS "Treat warnings as errors (GCC/Clang)" OFF)
option(PF_BUILD_FUZZ "Build libFuzzer targets (requires Clang)" OFF)
set(PF_SANITIZE "" CACHE STRING "Sanitizers, e.g. address,undefined (GCC/Clang)")

# Common warning / sanitizer flags applied to all pf targets via pf_options.
add_library(pf_options INTERFACE)
if(MSVC)
  target_compile_options(pf_options INTERFACE /W4)
else()
  target_compile_options(pf_options INTERFACE -Wall -Wextra -Wpedantic)
  if(PF_WARNINGS_AS_ERRORS)
    target_compile_options(pf_options INTERFACE -Werror)
  endif()
  if(PF_SANITIZE)
    target_compile_options(pf_options INTERFACE -fsanitize=${PF_SANITIZE} -fno-omit-frame-pointer)
    target_link_options(pf_options INTERFACE -fsanitize=${PF_SANITIZE})
  endif()
endif()

# OpenSSL-backed crypto: AUTO builds it when OpenSSL >= 3.0 is found, ON requires it, OFF skips it.
# Everything except core/src/crypto/ is OpenSSL-free (blocks only see the AeadProvider interface).
set(PF_WITH_OPENSSL "AUTO" CACHE STRING "OpenSSL-backed crypto: AUTO, ON or OFF")
set(PF_HAVE_OPENSSL OFF)
if(NOT PF_WITH_OPENSSL STREQUAL "OFF")
  if(PF_WITH_OPENSSL STREQUAL "ON")
    find_package(OpenSSL 3.0 REQUIRED)
  else()
    find_package(OpenSSL 3.0 QUIET)
  endif()
  if(OPENSSL_FOUND)
    set(PF_HAVE_OPENSSL ON)
  endif()
endif()
message(STATUS "pf: OpenSSL crypto = ${PF_HAVE_OPENSSL}")

file(GLOB PF_CORE_SRCS CONFIGURE_DEPENDS core/src/*.cpp core/src/*/*.cpp)
list(FILTER PF_CORE_SRCS EXCLUDE REGEX "/core/src/crypto/")
if(PF_HAVE_OPENSSL)
  file(GLOB PF_CRYPTO_SRCS CONFIGURE_DEPENDS core/src/crypto/*.cpp)
  list(APPEND PF_CORE_SRCS ${PF_CRYPTO_SRCS})
endif()
add_library(pf_core STATIC ${PF_CORE_SRCS})
target_include_directories(pf_core PUBLIC core/include)
target_link_libraries(pf_core PUBLIC pf_options)
if(PF_HAVE_OPENSSL)
  target_link_libraries(pf_core PUBLIC OpenSSL::SSL OpenSSL::Crypto)
  target_compile_definitions(pf_core PUBLIC PF_WITH_OPENSSL=1)
endif()

if(PF_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
